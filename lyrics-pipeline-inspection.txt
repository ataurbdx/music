/**
 * Fetches page HTML with Capacitor native HTTP or server preview proxy
 */
export async function fetchLyricsPageHtml(url: string): Promise<string> {
  if (!url || !url.startsWith('http')) {
    throw new Error('Valid HTTP/HTTPS URL is required.');
  }

  // Check Capacitor native platform
  const isNative = typeof window !== 'undefined' && (window as any).Capacitor?.isNativePlatform?.();

  if (isNative && (window as any).Capacitor?.Plugins?.CapacitorHttp) {
    const response = await (window as any).Capacitor.Plugins.CapacitorHttp.get({
      url,
      headers: {
        'User-Agent': 'Mozilla/5.0 (Linux; Android 14) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0 Mobile Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml',
      },
      readTimeout: 10000,
      connectTimeout: 10000,
    });
    return response.data;
  }

  // In Web / PWA preview mode, use the server proxy to bypass CORS
  const proxyUrl = `/api/proxy-fetch?url=${encodeURIComponent(url)}`;
  const res = await fetch(proxyUrl);
  if (!res.ok) {
    throw new Error(`Failed to fetch page (HTTP ${res.status})`);
  }
  return res.text();
}

/**
 * HTML text extractor and cleaner
 * Parses HTML with DOMParser, strips scripts, styles, nav, footers, ads,
 * and extracts clean plain text while maintaining block line breaks.
 */
export function extractCleanTextFromHtml(html: string, maxChars = 40000): string {
  if (!html) return '';

  if (typeof DOMParser === 'undefined') {
    // Basic regex fallback if DOMParser is unavailable
    return html
      .replace(/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/gi, '')
      .replace(/<style\b[^<]*(?:(?!<\/style>)<[^<]*)*<\/style>/gi, '')
      .replace(/<[^>]+>/g, ' ')
      .replace(/\s+/g, ' ')
      .trim()
      .slice(0, maxChars);
  }

  const parser = new DOMParser();
  const doc = parser.parseFromString(html, 'text/html');

  // Remove irrelevant elements while preserving article headers with song titles
  const tagsToRemove = [
    'script',
    'style',
    'noscript',
    'nav',
    'header.site-header',
    '#masthead',
    'footer.site-footer',
    '#colophon',
    'aside',
    'form',
    'iframe',
    'svg',
    '.ad',
    '.ads',
    '.sidebar',
    '.advertisement',
    '.comments',
    '#comments',
  ];

  tagsToRemove.forEach((sel) => {
    try {
      doc.querySelectorAll(sel).forEach((el) => el.remove());
    } catch {
      // Ignored
    }
  });

  // Convert block elements to newlines
  const blockTags = ['p', 'div', 'br', 'li', 'h1', 'h2', 'h3', 'h4', 'h5', 'h6', 'tr'];
  blockTags.forEach((tag) => {
    doc.querySelectorAll(tag).forEach((el) => {
      el.insertAdjacentText('afterend', '\n');
    });
  });

  const rawText = doc.body?.textContent || '';
  
  // Clean whitespace: collapse multiple horizontal spaces, keep clean newlines
  const lines = rawText
    .split('\n')
    .map((l) => l.replace(/[ \t\r]+/g, ' ').trim())
    .filter(Boolean);

  const cleanText = lines.join('\n');
  return cleanText.slice(0, maxChars);
}

import { APPROVED_LYRICS_SOURCES } from './sources.config';
import { fetchLyricsPageHtml } from './fetchPage';
import { normalizeBengali, normalizeLatin } from '../lib/normalize';

export interface CandidateLyricsUrl {
  url: string;
  siteId: string;
  siteName: string;
  pageTitle: string;
  snippet?: string;
  score: number;
}

/**
 * Searches approved sites and live search engine for candidates matching the song title
 */
export async function discoverLyricsCandidates(
  songTitle: string,
  enabledSiteIds: string[] = ['lyrics71', 'bangla-lyrics', 'chordzone', 'smule'],
  titleLatin?: string,
  titleBn?: string,
  language: 'bn' | 'en' = 'bn'
): Promise<CandidateLyricsUrl[]> {
  const candidates: CandidateLyricsUrl[] = [];
  const seenUrls = new Set<string>();

  const targetBn = titleBn || (normalizeBengali(songTitle) ? songTitle : '');
  const targetLatin = titleLatin || songTitle;

  const normTitleBn = normalizeBengali(targetBn);
  const normTitleLatin = normalizeLatin(targetLatin);

  // Extract search tokens for matching (min 2 chars)
  const bnTokens = normTitleBn.split(/\s+/).filter((t) => t.length >= 2);
  const latinTokens = normTitleLatin.split(/\s+/).filter((t) => t.length >= 2);

  // Helper to score a candidate title and snippet against song query
  const scoreCandidate = (candTitle: string, snippet = ''): number => {
    if (!candTitle || !candTitle.trim()) return 0;
    
    // Ignore generic navigation/boilerplates
    if (/^(home|read more|next|previous|older|newer|comments?|share|contact|about|privacy)/i.test(candTitle.trim())) {
      return 0;
    }

    const normCandBn = normalizeBengali(candTitle);
    const normCandLatin = normalizeLatin(candTitle);
    const normSnipBn = normalizeBengali(snippet);
    const normSnipLatin = normalizeLatin(snippet);

    let score = 0;

    // Exact or strong substring match in title
    if (normTitleBn && normCandBn) {
      if (normCandBn.includes(normTitleBn)) score = Math.max(score, 90);
      else if (normTitleBn.includes(normCandBn) && normCandBn.length >= normTitleBn.length * 0.7) score = Math.max(score, 85);
    }
    if (normTitleLatin && normCandLatin) {
      if (normCandLatin.includes(normTitleLatin)) score = Math.max(score, 90);
      else if (normTitleLatin.includes(normCandLatin) && normCandLatin.length >= normTitleLatin.length * 0.7) score = Math.max(score, 85);
    }

    // Token match count in title
    let matchedTokens = 0;
    if (bnTokens.length > 0) {
      for (const tok of bnTokens) {
        if (normCandBn.includes(tok)) matchedTokens++;
      }
      const ratio = matchedTokens / bnTokens.length;
      if (ratio >= 0.5 || matchedTokens >= 2) {
        score = Math.max(score, Math.round(ratio * 80));
      }
    }

    let matchedLatinTokens = 0;
    if (latinTokens.length > 0) {
      for (const tok of latinTokens) {
        if (normCandLatin.includes(tok)) matchedLatinTokens++;
      }
      const ratio = matchedLatinTokens / latinTokens.length;
      if (ratio >= 0.5 || matchedLatinTokens >= 2) {
        score = Math.max(score, Math.round(ratio * 80));
      }
    }

    // Snippet bonus
    if (snippet && ((normTitleBn && normSnipBn.includes(normTitleBn)) || (normTitleLatin && normSnipLatin.includes(normTitleLatin)))) {
      score = Math.min(100, score + 15);
    }

    return score;
  };

  // 1. Live Web Search via /api/search-lyrics
  try {
    const params = new URLSearchParams();
    params.set('q', songTitle);
    if (titleLatin) params.set('titleLatin', titleLatin);
    if (titleBn) params.set('titleBn', titleBn);
    params.set('language', language);

    const searchRes = await fetch(`/api/search-lyrics?${params.toString()}`);
    if (searchRes.ok) {
      const data = await searchRes.json();
      if (Array.isArray(data.results)) {
        for (const item of data.results) {
          if (!item.url || seenUrls.has(item.url)) continue;

          const localScore = scoreCandidate(item.title, item.snippet);
          const finalScore = Math.max(localScore, item.score || 0);

          // Strictly filter out results that do not reasonably match
          if (finalScore >= 40) {
            seenUrls.add(item.url);
            candidates.push({
              url: item.url,
              siteId: item.source || 'web',
              siteName: item.source === 'lyrics71.net' ? 'Lyrics 71' : item.source === 'bangla-lyrics.com' ? 'Bangla-Lyrics' : item.source === 'chordzone.org' ? 'ChordZone' : item.source || 'Web Archive',
              pageTitle: item.title || songTitle,
              snippet: item.snippet || '',
              score: finalScore,
            });
          }
        }
      }
    }
  } catch {
    // Web search optional fallback
  }

  // 2. Direct search on configured active sites if needed
  if (candidates.length < 4) {
    const activeSources = APPROVED_LYRICS_SOURCES.filter((s) => enabledSiteIds.includes(s.id));

    for (const source of activeSources) {
      try {
        const searchUrl = source.searchUrlTemplate.replace('{query}', encodeURIComponent(songTitle));
        const html = await fetchLyricsPageHtml(searchUrl);

        const parser = new DOMParser();
        const doc = parser.parseFromString(html, 'text/html');

        // Look for headings with links first (standard article listings)
        const headingAnchors = Array.from(doc.querySelectorAll('h1 a, h2 a, h3 a, article a.read-more-btn'));
        const allAnchors = headingAnchors.length > 0 ? headingAnchors : Array.from(doc.querySelectorAll('a[href]'));

        for (const a of allAnchors) {
          const href = a.getAttribute('href') || '';
          const linkText = (a.textContent || '').trim();

          // Reject empty text or generic read-more text
          if (!linkText || linkText.length < 2) continue;
          if (/^(read more|older|newer|home|share|next|previous)/i.test(linkText)) continue;

          if (!href.startsWith('http') && !href.startsWith('/')) continue;
          const fullUrl = href.startsWith('http') ? href : `https://${source.domain}${href}`;

          if (!fullUrl.includes(source.domain)) continue;
          if (/(\/category\/|\/tag\/|\/page\/|\/author\/|#|\/feed\/|\/comments\/)/i.test(fullUrl)) continue;
          if (fullUrl === `https://${source.domain}/` || fullUrl === `http://${source.domain}/`) continue;
          if (seenUrls.has(fullUrl)) continue;

          const score = scoreCandidate(linkText);

          // Require reasonable score threshold to avoid unrelated songs
          if (score >= 40) {
            seenUrls.add(fullUrl);
            candidates.push({
              url: fullUrl,
              siteId: source.id,
              siteName: source.name,
              pageTitle: linkText,
              score,
            });
          }
        }
      } catch {
        // Best effort per site
      }
    }
  }

  // Sort candidates by score descending
  return candidates.sort((a, b) => b.score - a.score).slice(0, 8);
}

import { normalizeBengali, computeBengaliRatio } from '../lib/normalize';
import { VerificationResult, LineMatchResult } from '../types';

/**
 * Calculates Levenshtein-based similarity between two short strings [0.0 - 1.0]
 */
function calculateStringSimilarity(s1: string, s2: string): number {
  if (s1 === s2) return 1.0;
  if (!s1.length || !s2.length) return 0.0;

  // If one contains the other entirely
  if (s1.includes(s2) || s2.includes(s1)) {
    const minLen = Math.min(s1.length, s2.length);
    const maxLen = Math.max(s1.length, s2.length);
    return minLen / maxLen >= 0.85 ? 0.95 : 0.90;
  }

  const longer = s1.length > s2.length ? s1 : s2;
  const shorter = s1.length > s2.length ? s2 : s1;

  // Optimized distance calculation for performance
  const costs: number[] = [];
  for (let i = 0; i <= longer.length; i++) {
    costs[i] = i;
  }

  for (let j = 1; j <= shorter.length; j++) {
    let nw = costs[0];
    costs[0] = j;
    for (let i = 1; i <= longer.length; i++) {
      const cj = costs[i];
      if (longer[i - 1] === shorter[j - 1]) {
        costs[i] = nw;
      } else {
        costs[i] = 1 + Math.min(costs[i], costs[i - 1], nw);
      }
      nw = cj;
    }
  }

  const distance = costs[longer.length];
  return (longer.length - distance) / longer.length;
}

/**
 * Deterministically verifies extracted Bengali lyrics against untrusted source page/input text.
 * Proves that zero lines were hallucinated or fabricated by the AI model.
 */
export function verifyExtractedLyrics(
  extractedLyrics: string,
  rawSourceText: string,
  threshold = 0.90,
  allowEnglish = false
): VerificationResult {
  const reasons: string[] = [];

  const rawLines = extractedLyrics
    .split('\n')
    .map((l) => l.trim())
    .filter(Boolean);

  if (rawLines.length === 0) {
    return {
      passed: false,
      needsAttention: true,
      coverage: 0,
      bengaliRatio: 0,
      language: allowEnglish ? 'en' : 'bn',
      lineCount: 0,
      matchedLinesCount: 0,
      lineDetails: [],
      reasons: ['No lyrics lines found in extraction output'],
    };
  }

  const bengaliRatio = computeBengaliRatio(extractedLyrics);
  const isEnglishLyrics = allowEnglish || bengaliRatio < 0.20;

  if (bengaliRatio < 0.40 && !allowEnglish) {
    reasons.push(
      `Low Bengali character ratio: ${(bengaliRatio * 100).toFixed(1)}% (minimum 40% required). If this is an English song, approve English lyrics.`
    );
  }

  if (rawLines.length < 4) {
    reasons.push(`Fewer than 4 lines extracted (${rawLines.length} lines). Song appears truncated.`);
  }

  const cleanNormalize = (text: string) => {
    if (isEnglishLyrics) {
      return text
        .normalize('NFC')
        .toLowerCase()
        .replace(/[!?,:;"'(){}\[\]<>~`@#$%^&*_\-+=/\\|।॥]/g, ' ')
        .replace(/\s+/g, ' ')
        .trim();
    }
    return normalizeBengali(text);
  };

  const normalizedSource = cleanNormalize(rawSourceText);
  const sourceLines = normalizedSource
    .split('\n')
    .map((l) => l.trim())
    .filter(Boolean);

  const lineDetails: LineMatchResult[] = [];
  let matchedCount = 0;

  for (let i = 0; i < rawLines.length; i++) {
    const originalLine = rawLines[i];
    const normLine = cleanNormalize(originalLine);
    const cleanedLine = normLine.replace(/\[[^\]]+\]/g, '').replace(/\([^\)]+\)/g, '').replace(/\s+/g, ' ').trim();

    if (!normLine || !cleanedLine) {
      lineDetails.push({
        lineIndex: i,
        text: originalLine,
        matched: true,
        bestSimilarity: 1.0,
      });
      matchedCount++;
      continue;
    }

    // Direct substring check in normalized source
    if (normalizedSource.includes(normLine) || normalizedSource.includes(cleanedLine)) {
      lineDetails.push({
        lineIndex: i,
        text: originalLine,
        matched: true,
        bestSimilarity: 1.0,
        matchedSourceSnippet: normLine,
      });
      matchedCount++;
      continue;
    }

    // Fuzzy check against each source line / chunk
    let bestSim = 0;
    let bestSnippet = '';

    for (const srcLine of sourceLines) {
      const cleanedSrc = srcLine.replace(/\[[^\]]+\]/g, '').replace(/\([^\)]+\)/g, '').replace(/\s+/g, ' ').trim();
      const sim = Math.max(
        calculateStringSimilarity(normLine, srcLine),
        calculateStringSimilarity(cleanedLine, cleanedSrc)
      );
      if (sim > bestSim) {
        bestSim = sim;
        bestSnippet = srcLine;
        if (bestSim >= 0.98) break;
      }
    }

    // If still not matched, check sliding sub-strings in source
    if (bestSim < threshold && cleanedLine.length > 8) {
      const lineLen = cleanedLine.length;
      for (let s = 0; s <= normalizedSource.length - lineLen; s += Math.max(1, Math.floor(lineLen / 3))) {
        const window = normalizedSource.slice(s, s + lineLen);
        const sim = calculateStringSimilarity(cleanedLine, window);
        if (sim > bestSim) {
          bestSim = sim;
          bestSnippet = window;
          if (bestSim >= threshold) break;
        }
      }
    }

    const matched = bestSim >= threshold;
    if (matched) {
      matchedCount++;
    }

    lineDetails.push({
      lineIndex: i,
      text: originalLine,
      matched,
      bestSimilarity: Number(bestSim.toFixed(3)),
      matchedSourceSnippet: bestSnippet || undefined,
    });
  }

  const coverage = Number((matchedCount / rawLines.length).toFixed(4));

  if (coverage < 0.98) {
    reasons.push(
      `Line verification coverage ${(coverage * 100).toFixed(1)}% is below 98%. ${
        rawLines.length - matchedCount
      } line(s) could not be verified in the source text.`
    );
  }

  const passed = coverage >= 0.98 && (allowEnglish || bengaliRatio >= 0.40) && rawLines.length >= 4;
  const needsAttention = !passed;

  return {
    passed,
    needsAttention,
    coverage,
    bengaliRatio,
    language: isEnglishLyrics ? 'en' : 'bn',
    lineCount: rawLines.length,
    matchedLinesCount: matchedCount,
    lineDetails,
    reasons,
  };
}

import { songRepository } from '../data/songRepository';
import { SongRecord, VerificationResult } from '../types';
import { discoverLyricsCandidates } from './discover';
import { fetchLyricsPageHtml } from './fetchPage';
import { extractCleanTextFromHtml } from './pageText';
import { geminiService } from '../services/gemini';
import { verifyExtractedLyrics } from './verify';

export interface QueueItem {
  songId: string;
  songTitle: string;
  status: 'pending' | 'processing' | 'completed' | 'needs_attention' | 'failed' | 'skipped';
  reason?: string;
  verification?: VerificationResult;
  extractedLyrics?: string;
  sourceUrl?: string;
}

export interface QueueState {
  isRunning: boolean;
  isPaused: boolean;
  currentIndex: number;
  items: QueueItem[];
  completedCount: number;
  failedCount: number;
}

type QueueListener = (state: QueueState) => void;

class LyricsQueueProcessor {
  private state: QueueState = {
    isRunning: false,
    isPaused: false,
    currentIndex: 0,
    items: [],
    completedCount: 0,
    failedCount: 0,
  };

  private listeners: Set<QueueListener> = new Set();
  private abortRequested = false;

  subscribe(listener: QueueListener): () => void {
    this.listeners.add(listener);
    listener(this.state);
    return () => this.listeners.delete(listener);
  }

  private notify(): void {
    for (const listener of this.listeners) {
      listener({ ...this.state, items: [...this.state.items] });
    }
  }

  getState(): QueueState {
    return { ...this.state, items: [...this.state.items] };
  }

  startQueue(songs: SongRecord[]): void {
    if (this.state.isRunning) return;

    this.abortRequested = false;
    this.state = {
      isRunning: true,
      isPaused: false,
      currentIndex: 0,
      items: songs.map((s) => ({
        songId: s.id,
        songTitle: s.titleBn || s.titleLatin,
        status: 'pending',
      })),
      completedCount: 0,
      failedCount: 0,
    };

    this.notify();
    this.processNext();
  }

  pause(): void {
    this.state.isPaused = true;
    this.notify();
  }

  resume(): void {
    if (!this.state.isRunning) return;
    this.state.isPaused = false;
    this.notify();
    this.processNext();
  }

  cancel(): void {
    this.abortRequested = true;
    this.state.isRunning = false;
    this.state.isPaused = false;
    this.notify();
  }

  private async processNext(): Promise<void> {
    if (this.abortRequested || this.state.isPaused || !this.state.isRunning) return;

    if (this.state.currentIndex >= this.state.items.length) {
      this.state.isRunning = false;
      this.notify();
      return;
    }

    const item = this.state.items[this.state.currentIndex];
    item.status = 'processing';
    this.notify();

    try {
      const song = await songRepository.getSongById(item.songId);
      if (!song) {
        item.status = 'failed';
        item.reason = 'Song not found in library';
        this.state.failedCount++;
      } else {
        await this.processSingleSong(song, item);
      }
    } catch (err: any) {
      if (err?.status === 429) {
        // Quota exhausted backoff / pause
        item.status = 'failed';
        item.reason = 'Quota limit reached (429). Queue paused.';
        this.state.failedCount++;
        this.pause();
        return;
      }

      item.status = 'failed';
      item.reason = err?.message || 'Processing error';
      this.state.failedCount++;
    }

    this.state.currentIndex++;
    this.notify();

    // Small delay between requests to be polite to sites
    setTimeout(() => {
      this.processNext();
    }, 1200);
  }

  private async processSingleSong(song: SongRecord, item: QueueItem): Promise<void> {
    // 1. Discover candidates from approved sites
    const searchTitle = song.titleBn || song.titleLatin;
    const candidates = await discoverLyricsCandidates(searchTitle);

    if (candidates.length === 0) {
      item.status = 'needs_attention';
      item.reason = 'No candidates found on approved sites. Try Paste and Clean.';
      this.state.failedCount++;
      return;
    }

    // Try top candidate
    const topCandidate = candidates[0];
    const html = await fetchLyricsPageHtml(topCandidate.url);
    const cleanPageText = extractCleanTextFromHtml(html);

    // 2. Extract with Gemini
    const extractRes = await geminiService.extractLyrics(cleanPageText, searchTitle);

    if (!extractRes.found || !extractRes.lyricsBn) {
      item.status = 'needs_attention';
      item.reason = 'Lyrics not found in candidate page content.';
      this.state.failedCount++;
      return;
    }

    // 3. Verify in code
    const verification = verifyExtractedLyrics(extractRes.lyricsBn, cleanPageText);
    item.verification = verification;
    item.extractedLyrics = extractRes.lyricsBn;
    item.sourceUrl = topCandidate.url;

    if (verification.passed) {
      // Auto-save as unreviewed if song has missing lyrics
      if (song.lyricsStatus === 'missing') {
        const now = new Date().toISOString();
        await songRepository.updateSong(song.id, {
          lyricsBn: extractRes.lyricsBn,
          lyricsStatus: 'unreviewed',
          lyricsOrigin: 'fetched',
          sources: [
            {
              url: topCandidate.url,
              siteId: topCandidate.siteId,
              fetchedAt: now,
              coverage: verification.coverage,
            },
          ],
        });
        item.status = 'completed';
        this.state.completedCount++;
      } else {
        // Do not silently overwrite existing lyrics
        item.status = 'needs_attention';
        item.reason = 'Song already has lyrics; user review required to overwrite.';
      }
    } else {
      item.status = 'needs_attention';
      item.reason = verification.reasons.join('; ');
      this.state.failedCount++;
    }
  }
}

interface AICallOptions {
  apiKey?: string;
  provider?: 'gemini' | 'openai_compatible';
  model?: string;
  baseUrl?: string;
  systemInstruction?: string;
  prompt: string;
  responseMimeType?: 'application/json' | 'text/plain';
  timeoutMs?: number;
}

async function callAIModel(opts: AICallOptions): Promise<{ text: string; modelUsed: string; providerUsed: string }> {
  const provider = opts.provider || (opts.baseUrl || opts.apiKey?.startsWith('cmd-') || opts.apiKey?.startsWith('sk-') ? 'openai_compatible' : 'gemini');
  const timeoutMs = opts.timeoutMs || 12000;

  if (provider === 'openai_compatible') {
    // CommandCode AI or OpenAI-compatible endpoint
    const apiKey = (opts.apiKey && opts.apiKey.trim()) || process.env.COMMANDCODE_API_KEY || process.env.OPENAI_API_KEY || '';
    if (!apiKey) {
      throw new Error('NO_API_KEY: Please provide your CommandCode / OpenAI-compatible API key in Settings.');
    }

    const rawBaseUrl = (opts.baseUrl && opts.baseUrl.trim()) || 'https://api.commandcode.ai/v1';
    const baseUrl = rawBaseUrl.replace(/\/+$/, '');
    const endpoint = `${baseUrl}/chat/completions`;
    const model = (opts.model && opts.model.trim()) || 'gpt-4o-mini';

    const messages: any[] = [];
    if (opts.systemInstruction) {
      messages.push({ role: 'system', content: opts.systemInstruction });
    }
    messages.push({ role: 'user', content: opts.prompt });

    const body: any = {
      model,
      messages,
      temperature: 0.1,
    };
    if (opts.responseMimeType === 'application/json') {
      body.response_format = { type: 'json_object' };
    }

    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeoutMs);

    try {
      const resp = await fetch(endpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${apiKey}`,
        },
        body: JSON.stringify(body),
        signal: controller.signal,
      });
      clearTimeout(timer);

      if (!resp.ok) {
        const errJson = await resp.json().catch(() => ({}));
        const errMsg = errJson?.error?.message || errJson?.message || `Custom AI provider returned error (HTTP ${resp.status})`;
        const err = new Error(errMsg);
        (err as any).status = resp.status;
        throw err;
      }

      const data = await resp.json();
      const content = data?.choices?.[0]?.message?.content || '';
      return { text: content, modelUsed: model, providerUsed: 'openai_compatible' };
    } catch (e: any) {
      clearTimeout(timer);
      if (e.name === 'AbortError') {
        throw new Error(`AI request to ${model} timed out after ${Math.round(timeoutMs / 1000)}s.`);
      }
      throw e;
    }
  }

  // Google Gemini AI Studio provider
  const apiKey = (opts.apiKey && opts.apiKey.trim()) || process.env.GEMINI_API_KEY || '';
  if (!apiKey) {
    throw new Error('NO_API_KEY: Please provide your Google AI Studio key in Settings.');
  }

  const ai = new GoogleGenAI({
    apiKey,
    httpOptions: {
      headers: {
        'User-Agent': 'aistudio-build',
      },
    },
  });

  const preferredModel = (opts.model && opts.model.trim()) || 'gemini-3.5-flash-lite';
  const fallbackModels = ['gemini-3.5-flash-lite', 'gemini-flash-lite-latest', 'gemini-3.5-flash'];
  const modelsToTry = [preferredModel, ...fallbackModels.filter(m => m !== preferredModel)];

  let lastError: any = null;
  for (const modelName of modelsToTry) {
    try {
      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(), Math.min(timeoutMs, 9000));

      const generatePromise = ai.models.generateContent({
        model: modelName,
        contents: opts.prompt,
        config: {
          systemInstruction: opts.systemInstruction,
          temperature: 0.1,
          responseMimeType: opts.responseMimeType,
        },
      });

      const response = await Promise.race([
        generatePromise,
        new Promise((_, reject) =>
          controller.signal.addEventListener('abort', () =>
            reject(new Error(`Timeout waiting for model ${modelName}`))
          )
        ),
      ]) as any;

      clearTimeout(timer);
      const text = response?.text || '';
      return { text, modelUsed: modelName, providerUsed: 'gemini' };
    } catch (err: any) {
      lastError = err;
      if (err?.status === 401 || err?.status === 403) {
        throw err;
      }
    }
  }

  throw lastError || new Error('All Gemini models failed or timed out.');
}

// Helper for title normalization and stop words
function normalizeSearchText(text: string): string {
  if (!text) return '';
  return text
    .toLowerCase()
    .replace(/[^\p{L}\p{N}\s]/gu, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

const COMMON_BANGLA_STOPWORDS = new Set([
  'lyrics', 'song', 'gaan', 'by', 'mp3', 'video', 'chords', 'লিরিক্স', 'গান',
  'the', 'a', 'an', 'in', 'on', 'of', 'and', 'er', 'te', 'ke', 'amra', 'tumi', 'ami',
  'bengali', 'bangla', 'album', 'full', 'official', 'audio'
]);

function calculateTitleMatchScore(query: string, candidateTitle: string): number {
  if (!query || !candidateTitle) return 0;
  const qNorm = normalizeSearchText(query);
  const cNorm = normalizeSearchText(candidateTitle);

  if (!qNorm || !cNorm) return 0;

  // Exact match
  if (cNorm === qNorm) {
    return 100;
  }

  // Candidate title contains the exact multi-word query phrase
  if (cNorm.includes(qNorm)) {
    return 95;
  }

  // Query contains candidate ONLY IF candidate is substantial (at least 70% of query length)
  if (qNorm.includes(cNorm) && cNorm.length >= qNorm.length * 0.7) {
    return 85;
  }

  const qTokens = qNorm.split(/\s+/).filter((t) => t.length >= 2 && !COMMON_BANGLA_STOPWORDS.has(t));
  const cTokens = new Set(cNorm.split(/\s+/).filter((t) => t.length >= 2));

  if (qTokens.length === 0) return 0;

  let matched = 0;
  for (const t of qTokens) {
    if (cTokens.has(t)) {
      matched += 1;
    } else {
      for (const ct of cTokens) {
        if (ct.length >= 3 && (ct.startsWith(t) || t.startsWith(ct))) {
          matched += 0.8;
          break;
        }
      }
    }
  }

  const tokenScore = Math.round((matched / qTokens.length) * 100);

  // If less than half the query tokens matched, it's not the same song!
  if (matched < Math.ceil(qTokens.length * 0.5)) {
    return Math.min(tokenScore, 30);
  }

  return tokenScore;
}

// 6. Safe web proxy for fetching any webpage (bypasses browser CORS)
app.get('/api/proxy-fetch', async (req, res) => {
  try {
    const targetUrl = req.query.url as string;
    if (!targetUrl) {
      return res.status(400).json({ error: 'URL is required' });
    }

    const parsedUrl = new URL(targetUrl);
    if (!['http:', 'https:'].includes(parsedUrl.protocol)) {
      return res.status(400).json({ error: 'Only http and https protocols supported' });
    }

    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), 12000);

    const response = await fetch(targetUrl, {
      signal: controller.signal,
      redirect: 'follow',
      headers: {
        'User-Agent': 'Mozilla/5.0 (Linux; Android 14; Mobile; Pixel 7) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0 Mobile Safari/537.36',
        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8',
        'Accept-Language': 'bn,en-US,en;q=0.9',
        'Cache-Control': 'no-cache',
        'Pragma': 'no-cache',
      },
    });
    clearTimeout(timeout);

    if (!response.ok) {
      return res.status(response.status).json({ error: `Fetch failed with status ${response.status} (${response.statusText})` });
    }

    const text = await response.text();
    res.setHeader('Content-Type', 'text/html; charset=utf-8');
    res.send(text);
  } catch (err: any) {
    res.status(500).json({ error: err?.message || String(err) });
  }
});

// Helper to sanitize YouTube URLs and extract canonical watch URL
function cleanYouTubeUrl(inputUrl: string): string {
  if (!inputUrl) return '';
  const trimmed = inputUrl.trim();
  const match = trimmed.match(/(?:youtu\.be\/|youtube\.com\/(?:embed\/|v\/|watch\?v=|watch\?.+&v=))([\w-]{11})/i);
  if (match && match[1]) {
    return `https://www.youtube.com/watch?v=${match[1]}`;
  }
  return trimmed;
}

function findYtDlp(): string {
  if (process.env.YTDLP_PATH && fs.existsSync(process.env.YTDLP_PATH)) {
    return process.env.YTDLP_PATH;
  }
  const possiblePaths = [
    path.resolve(__dirname, 'bin', process.platform === 'win32' ? 'yt-dlp.exe' : 'yt-dlp'),
    path.resolve(__dirname, 'bin', 'yt-dlp'),
    'E:\\tools\\YouTube Playlist Downloader\\yt-dlp.exe',
  ];
  for (const p of possiblePaths) {
    if (fs.existsSync(p)) return p;
  }
  return process.platform === 'win32' ? 'yt-dlp.exe' : 'yt-dlp';
}

function findFfmpeg(): string | null {

    '/usr/bin/ffmpeg',
    '/usr/local/bin/ffmpeg',
  ];
  for (const p of possiblePaths) {
    if (fs.existsSync(p)) return p;
  }
  return null;
}

  if (!url) {
    return res.status(400).json({ error: 'url is required' });
  }

  const cleanUrl = cleanYouTubeUrl(url);

  let responded = false;
  const sendJson = (status: number, data: any) => {
    if (responded || res.headersSent) return;
    responded = true;
    res.status(status).json(data);
  };

  try {
    const ytdlpArgs = [
      '--dump-single-json',
      '--no-playlist',
      '--no-warnings',
      '--no-check-certificates',
      '--user-agent',
      'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
      '--extractor-args',
      'youtube:player_client=android,web',
    ];

    if (FFMPEG_BIN) {
      ytdlpArgs.push('--ffmpeg-location', FFMPEG_BIN);
    }
    if (COOKIES_FILE) {
      ytdlpArgs.push('--cookies', COOKIES_FILE);
    }
    if (fs.existsSync(process.execPath)) {
      ytdlpArgs.push('--js-runtimes', `node:${process.execPath}`);
    }

    ytdlpArgs.push(cleanUrl);

    const ytdlp = spawn(YTDLP_BIN, ytdlpArgs);

    let stdout = '';
    let stderr = '';

    const timeout = setTimeout(() => {
      try { ytdlp.kill('SIGTERM'); } catch {}
      sendJson(504, { error: 'YouTube metadata extraction timed out.' });
    }, 30000);

    ytdlp.stdout?.on('data', (d) => {
      stdout += d.toString();
    });
    ytdlp.stderr?.on('data', (d) => {
      stderr += d.toString();
    });

    ytdlp.on('error', (err) => {
      clearTimeout(timeout);
      sendJson(500, { error: `Process execution error: ${err.message}` });
    });

    ytdlp.on('close', (code) => {
      clearTimeout(timeout);
      if (code !== 0) {
        return sendJson(500, {
          error: stderr.replace(/WARNING:.*\n?/g, '').trim() || 'Failed to extract video information from YouTube.',
        });

      try {
        const info = JSON.parse(stdout);
        sendJson(200, {
          id: info.id,
          title: info.title,
          duration: info.duration,
          thumbnail: info.thumbnail,
          uploader: info.uploader,
        });
      } catch {
        sendJson(500, { error: 'Could not parse video metadata.' });
      }
    });
  } catch (err: any) {
    sendJson(500, { error: err?.message || 'Server error' });
  }
});

app.all(['/api/playlist', '/api/audio/playlist'], async (req, res) => {
  const url = (req.body?.url || (req.query?.url as string) || '').trim();
  if (!url) {
    return res.status(400).json({ error: 'url is required' });
  }

  let responded = false;
  const sendJson = (status: number, data: any) => {
    if (responded || res.headersSent) return;
    responded = true;
    res.status(status).json(data);
  };

  try {
    const ytdlpArgs = [
      '--dump-single-json',
      '--flat-playlist',
      '--skip-download',
      '--no-warnings',
      '--no-check-certificates',
      '--user-agent',
      'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
      '--extractor-args',
      'youtube:player_client=android,web',
    ];

    if (FFMPEG_BIN) {
      ytdlpArgs.push('--ffmpeg-location', FFMPEG_BIN);
    }
    if (COOKIES_FILE) {
      ytdlpArgs.push('--cookies', COOKIES_FILE);
    }
    if (fs.existsSync(process.execPath)) {
      ytdlpArgs.push('--js-runtimes', `node:${process.execPath}`);
    }

    ytdlpArgs.push(url);

    const ytdlp = spawn(YTDLP_BIN, ytdlpArgs);

    let stdout = '';
    let stderr = '';

    const timeout = setTimeout(() => {
      try { ytdlp.kill('SIGTERM'); } catch {}
      sendJson(504, { error: 'Playlist extraction timed out.' });
    }, 45000);

    ytdlp.stdout?.on('data', (d) => {
      stdout += d.toString();
    });
    ytdlp.stderr?.on('data', (d) => {
      stderr += d.toString();
    });

    ytdlp.on('error', (err) => {
      clearTimeout(timeout);
      sendJson(500, { error: `yt-dlp execution error: ${err.message}` });
    });

    ytdlp.on('close', (code) => {
      clearTimeout(timeout);
      if (code !== 0 && !stdout) {
        return sendJson(500, {
          error: stderr.replace(/WARNING:.*\n?/g, '').trim() || 'Failed to extract playlist information.',
        });
      }
      try {
        const info = JSON.parse(stdout);
        const entries = (info.entries || []).map((e: any, idx: number) => {
          const vidId = e.id || e.videoId;
          let thumb = '';
          if (Array.isArray(e.thumbnails) && e.thumbnails.length > 0) {
            thumb = e.thumbnails[e.thumbnails.length - 1]?.url || e.thumbnails[0]?.url;
          } else if (e.thumbnail) {
            thumb = e.thumbnail;
          } else if (vidId) {
            thumb = `https://img.youtube.com/vi/${vidId}/hqdefault.jpg`;
          }

          return {
            id: vidId,
            videoId: vidId,
            title: e.title || `Track ${idx + 1}`,
            duration: e.duration || 0,
            thumbnail: thumb,
            thumbnailUrl: thumb,
            url: e.url || (vidId ? `https://www.youtube.com/watch?v=${vidId}` : ''),
            youtubeUrl: e.url || (vidId ? `https://www.youtube.com/watch?v=${vidId}` : ''),
            uploader: e.uploader || e.channel || info.uploader || 'YouTube',
            channelTitle: e.uploader || e.channel || info.uploader || 'YouTube',
            position: idx,
          };
        }).filter((e: any) => Boolean(e.videoId) && e.title !== '[Private video]' && e.title !== '[Deleted video]');

        let playlistThumb = '';
        if (Array.isArray(info.thumbnails) && info.thumbnails.length > 0) {
          playlistThumb = info.thumbnails[info.thumbnails.length - 1]?.url || info.thumbnails[0]?.url;
        } else if (info.thumbnail) {
          playlistThumb = info.thumbnail;
        } else if (entries[0]?.thumbnailUrl) {
          playlistThumb = entries[0].thumbnailUrl;
        }

        sendJson(200, {
          id: info.id,
          title: info.title || 'YouTube Playlist',
          author: info.uploader || info.channel || 'YouTube Community',
          uploader: info.uploader || info.channel || 'YouTube Community',
          thumbnail: playlistThumb,
          totalCount: entries.length,
          entries,
          items: entries,
        });
      } catch (err: any) {
        sendJson(500, { error: 'Failed to parse playlist data: ' + err.message });
      }
    });
  } catch (err: any) {
    sendJson(500, { error: err?.message || 'Server error' });
  }
});

// Real YouTube audio extraction and conversion via yt-dlp + ffmpeg
app.get('/api/audio/download', async (req, res) => {
  const url = (req.query.url as string || '').trim();
  const format = ((req.query.format as string) || 'mp3').toLowerCase();
  const quality = ((req.query.quality as string) || '320k').toLowerCase().replace('kbps', 'k');

  if (!url) {
    return res.status(400).json({ error: 'YouTube URL parameter is required' });
  }

  const cleanUrl = cleanYouTubeUrl(url);

  const supportedFormats = ['mp3', 'm4a', 'opus', 'wav', 'flac'];
  const cleanFormat = supportedFormats.includes(format) ? format : 'mp3';
  const cleanQuality = ['320k', '256k', '192k', '128k'].includes(quality) ? quality : '320k';

  const mimeTypes: Record<string, string> = {
    mp3: 'audio/mpeg',
    m4a: 'audio/mp4',
    opus: 'audio/opus',
    wav: 'audio/wav',
    flac: 'audio/flac',
  };

  const tempId = `yt_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`;
  const tempDir = os.tmpdir();
  const tempPrefix = path.join(tempDir, tempId);

  const ytdlpArgs = [
    '--no-playlist',
    '--no-warnings',
    '--no-check-certificates',
    '--user-agent',
    'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36',
    '--extractor-args',
    'youtube:player_client=android,web',
    '--max-filesize',
    '65M',
    '-x',
    '--audio-format',
    cleanFormat,
    '--audio-quality',
    cleanQuality,
  ];

  if (FFMPEG_BIN) {
    ytdlpArgs.push('--ffmpeg-location', FFMPEG_BIN);
  }
  if (COOKIES_FILE) {
    ytdlpArgs.push('--cookies', COOKIES_FILE);
  }
  if (fs.existsSync(process.execPath)) {
    ytdlpArgs.push('--js-runtimes', `node:${process.execPath}`);
  }

  ytdlpArgs.push('-o', `${tempPrefix}.%(ext)s`, cleanUrl);

  let responded = false;
  const sendJson = (status: number, data: any) => {
    if (responded || res.headersSent) return;
    responded = true;
    res.status(status).json(data);
  };

  try {
    const child = spawn(YTDLP_BIN, ytdlpArgs);

    let stderr = '';

    const timeout = setTimeout(() => {
      try { child.kill('SIGTERM'); } catch {}
      sendJson(504, { error: 'Audio download timed out after 75 seconds.' });
    }, 75000);

    child.stderr?.on('data', (d) => {
      stderr += d.toString();
    });

    child.on('error', (err) => {
      clearTimeout(timeout);
      sendJson(500, { error: `Process execution error: ${err.message}` });
    });

    child.on('close', (code) => {
      clearTimeout(timeout);
      if (responded || res.headersSent) return;

      if (code !== 0) {
        const cleanErr = stderr
          .replace(/WARNING:.*\n?/g, '')
          .replace(/ERROR: \[youtube\] /g, '')
          .trim();
        return sendJson(500, {
          error: cleanErr || 'Failed to extract audio from YouTube. The video may be restricted or unavailable.',
        });
      }

      // Find the converted file
      const expectedFile = `${tempPrefix}.${cleanFormat}`;
      if (!fs.existsSync(expectedFile)) {
        // Try looking for any file starting with tempPrefix
        const dir = tempDir;
        const files = fs.readdirSync(dir).filter((f) => f.startsWith(tempId));
        if (files.length > 0) {
          const matchPath = path.join(dir, files[0]);
          responded = true;
          res.setHeader('Content-Type', mimeTypes[cleanFormat] || 'audio/mpeg');
          res.setHeader('Content-Disposition', `attachment; filename="tour_audio.${cleanFormat}"`);
          return res.sendFile(matchPath, () => {
            try { fs.unlinkSync(matchPath); } catch {}
          });
        }
        return sendJson(500, { error: 'Extracted audio file not found on server.' });
      }

      responded = true;
      res.setHeader('Content-Type', mimeTypes[cleanFormat] || 'audio/mpeg');
      res.setHeader('Content-Disposition', `attachment; filename="tour_audio.${cleanFormat}"`);
      res.sendFile(expectedFile, () => {
        try { fs.unlinkSync(expectedFile); } catch {}
      });
    });

    req.on('close', () => {
      clearTimeout(timeout);
      try { child.kill('SIGTERM'); } catch {}
    });
  } catch (err: any) {
    sendJson(500, { error: err?.message || 'Download handler exception' });
  }
});

// Start dev or production server
async function startServer() {
  if (process.env.NODE_ENV !== 'production') {
    const vite = await createViteServer({
      server: { middlewareMode: true },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    app.use(express.static(path.resolve(__dirname, 'dist')));
    app.get('*', (_req, res) => {
      res.sendFile(path.resolve(__dirname, 'dist', 'index.html'));
    });
  }

  app.listen(PORT, '0.0.0.0', () => {
    console.log(`Tour Lyrics server running at http://localhost:${PORT}`);
  });
}

startServer();
